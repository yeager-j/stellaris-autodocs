# log: consuming `@fwd` corrupts the file with a diagnostic that never names the constant —
# `unknown command 'tier' for MTTH/script value` — so the resolver has to notice the broken
# reference itself.

@fwd_target = 7
@cycle_a = @cycle_b
@cycle_b = @cycle_a
