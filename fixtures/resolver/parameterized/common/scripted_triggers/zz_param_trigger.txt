# Original work; no Stellaris file is reproduced.
#
# `trig_param` carries a `$COUNT$` parameter substitution nested inside `check_variable`, and
# `trig_root_param` carries one as its own top-level field key — the same open cell at both
# depths a definition can carry it. `trig_plain` is the control: a trigger with no parameter
# at all, so the same row still resolves a corpus that never exercises the open cell.

trig_param = {
	check_variable = {
		which = oracle_count
		value = $COUNT$
	}
}

trig_root_param = {
	$MODE$ = yes
}

trig_plain = {
	always = yes
}
